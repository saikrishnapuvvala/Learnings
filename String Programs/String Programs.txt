public class Main {

    // 1. Reverse String
    static String reverseString(String s) {
        String rev = "";
        for (int i = s.length() - 1; i >= 0; i--) {
            rev += s.charAt(i);
        }
        return rev;
    }

    // 2. Palindrome Check
    static boolean isPalindrome(String s) {
        return s.equals(reverseString(s));
    }

    // 3. Count vowels and consonants
    static void countVowelsConsonants(String s) {
        s = s.toLowerCase();
        int v = 0, c = 0;

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (ch >= 'a' && ch <= 'z') {
                if (ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u')
                    v++;
                else
                    c++;
            }
        }

        System.out.println("Vowels: " + v);
        System.out.println("Consonants: " + c);
    }

    // 4. Remove spaces
    static String removeSpaces(String s) {
        return s.replace(" ", "");
    }

    // 5. Toggle characters
    static String toggle(String s) {
        String res = "";

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (Character.isUpperCase(ch))
                res += Character.toLowerCase(ch);
            else
                res += Character.toUpperCase(ch);
        }

        return res;
    }

    // 6. Uppercase conversion
    static String toUpper(String s) {
        return s.toUpperCase();
    }

    // 7. Duplicate characters
    static void printDuplicates(String s) {
        System.out.print("Duplicates: ");

        for (int i = 0; i < s.length(); i++) {
            for (int j = i + 1; j < s.length(); j++) {
                if (s.charAt(i) == s.charAt(j)) {
                    System.out.print(s.charAt(i) + " ");
                    break;
                }
            }
        }
        System.out.println();
    }

    // 8. Frequency of characters
    static void frequency(String s) {
        int[] f = new int[256];

        for (int i = 0; i < s.length(); i++) {
            f[s.charAt(i)]++;
        }

        for (int i = 0; i < 256; i++) {
            if (f[i] > 0) {
                System.out.println((char) i + " -> " + f[i]);
            }
        }
    }

    // 9. Anagram check
    static boolean isAnagram(String s1, String s2) {

        if (s1.length() != s2.length()) return false;

        s1 = s1.toLowerCase();
        s2 = s2.toLowerCase();

        char[] a = s1.toCharArray();
        char[] b = s2.toCharArray();

        java.util.Arrays.sort(a);
        java.util.Arrays.sort(b);

        return java.util.Arrays.equals(a, b);
    }

    // 10. Longest word
    static String longestWord(String s) {
        String[] words = s.split(" ");
        String longest = "";

        for (String w : words) {
            if (w.length() > longest.length()) {
                longest = w;
            }
        }
        return longest;
    }

    // 11. Count words
    static int countWords(String s) {
        return s.split(" ").length;
    }

    // 12. Reverse sentence
    static String reverseSentence(String s) {
        String[] words = s.split(" ");
        String rev = "";

        for (int i = words.length - 1; i >= 0; i--) {
            rev += words[i] + " ";
        }

        return rev;
    }

    // 13. String compression
    static String compress(String str) {
        StringBuilder sb = new StringBuilder();
        int c = 1;

        for (int i = 0; i < str.length(); i++) {
            while (i < str.length() - 1 &&
                    str.charAt(i) == str.charAt(i + 1)) {
                c++;
                i++;
            }
            sb.append(str.charAt(i)).append(c);
            c = 1;
        }
        return sb.toString();
    }

    // 14. Check substring
    static boolean checkSubstring(String str, String sub) {
        return str.contains(sub);
    }

    // 15. Lexicographically largest string
    static String largestLexico(String[] arr) {
        String lar = arr[0];

        for (int i = 1; i < arr.length; i++) {
            if (arr[i].compareTo(lar) > 0) {
                lar = arr[i];
            }
        }
        return lar;
    }

    // 16. ASCII values
    static void asciiValues(String s) {
        for (int i = 0; i < s.length(); i++) {
            System.out.println(s.charAt(i) + " -> " + (int) s.charAt(i));
        }
    }

    // 17. Count special characters
    static int countSpecial(String s) {
        int c = 0;

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (!((ch >= 'A' && ch <= 'Z') ||
                  (ch >= 'a' && ch <= 'z') ||
                  (ch >= '0' && ch <= '9'))) {
                c++;
            }
        }
        return c;
    }

    // 18. Remove vowels
    static String removeVowels(String s) {
        String r = "";

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            if (!(ch == 'a' || ch == 'e' || ch == 'i' || ch == 'o' || ch == 'u' ||
                  ch == 'A' || ch == 'E' || ch == 'I' || ch == 'O' || ch == 'U')) {
                r += ch;
            }
        }
        return r;
    }

    // 19. First non-repeating character
    static char firstNonRepeating(String s) {

        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            int count = 0;

            for (int j = 0; j < s.length(); j++) {
                if (s.charAt(j) == c) {
                    count++;
                }
            }

            if (count == 1) {
                return c;
            }
        }
        return '-';
    }

    // MAIN METHOD (Testing all)
    public static void main(String[] args) {

        System.out.println("Reverse: " + reverseString("hello"));
        System.out.println("Palindrome: " + isPalindrome("madam"));

        countVowelsConsonants("haiibyee");

        System.out.println("Remove spaces: " + removeSpaces("ha ii by ee"));
        System.out.println("Toggle: " + toggle("haIadINE"));
        System.out.println("Uppercase: " + toUpper("asdfg"));

        printDuplicates("asdfghkjhgfds");

        System.out.println("Anagram: " + isAnagram("Sai", "ias"));

        System.out.println("Longest word: " +
                longestWord("I am Sai Krishna Puvvala"));

        System.out.println("Word count: " +
                countWords("I am Sai Krishna Puvvala"));

        System.out.println("Reverse sentence: " +
                reverseSentence("I am Sai Krishna"));

        System.out.println("Compressed: " + compress("aaabbbcc"));

        System.out.println("Substring: " +
                checkSubstring("aaabbbcc", "bbb"));

        String[] arr = {"apple", "main", "cat", "bin"};
        System.out.println("Largest lexicographic: " + largestLexico(arr));

        asciiValues("Sai");

        System.out.println("Special chars: " +
                countSpecial("Hello@123#World!$"));

        System.out.println("Remove vowels: " +
                removeVowels("Hii All"));

        System.out.println("First non-repeating: " +
                firstNonRepeating("aaabbbcccddds"));
    }
}
